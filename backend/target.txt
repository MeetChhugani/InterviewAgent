    # â”€â”€ Candidate answer handler â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
    async def handle_user_speech(msg: llm.ChatMessage):
        nonlocal is_processing_turn, interview_phase

        # We must be in PROCESSING phase (set by STOP ANSWERING).
        if interview_phase != "PROCESSING":
            logger.warning(
                "[FLOW] ANSWER PROCESSING BLOCKED reason=wrong_phase "
                "phase=%s current_q=%s", interview_phase, current_q,
            )
            is_processing_turn = False
            return

        try:
            text = getattr(msg, "text", getattr(msg, "content", str(msg)))
            if not isinstance(text, str):
                text = str(text)
            text = text.strip()

            # If LiveKit AgentSession committed the turn before STT was ready, text is empty.
            if not text and candidate_transcript_buffer:
                text = " ".join(candidate_transcript_buffer)
            
            if not text:
                logger.warning("[STT] FINAL TRANSCRIPT NOT READY")
                logger.info("[STT] WAITING FOR FINAL TRANSCRIPT")
                try:
                    await asyncio.wait_for(final_transcript_event.wait(), timeout=10.0)
                    text = " ".join(candidate_transcript_buffer).strip()
                except asyncio.TimeoutError:
                    logger.warning("[STT] FINAL TRANSCRIPT TIMEOUT")
                    interview_phase = "WAITING_FOR_ANSWER"
                    is_processing_turn = False
                    await publish({"type": "waiting_for_answer", "question_number": current_q, "question_count": MAX_INTERVIEW_QUESTIONS})
                    return

            if not text:
                logger.warning("[STT] Transcript empty after wait.")
                interview_phase = "WAITING_FOR_ANSWER"
                is_processing_turn = False
                await publish({"type": "waiting_for_answer", "question_number": current_q, "question_count": MAX_INTERVIEW_QUESTIONS})
                return

            logger.info("[STT] FINAL ANSWER RECEIVED:\n%s", text)
            
            interview_phase = "EVALUATING_ANSWER"
            logger.info("[FLOW] STARTING ANSWER EVALUATION current_q=%s", current_q)
            
            eval_data = await evaluate_answer(current_question_text, text)

            result = {
                "question_number":  current_q,
                "question":         current_question_text,
                "candidate_answer": text,
                "answer":           text,   # frontend compatibility
                "evaluation":       eval_data,
                "score":            eval_data.get("score", 0),
                "correctness":      eval_data.get("correctness", ""),
                "feedback":         eval_data.get("feedback", ""),
                "strengths":        eval_data.get("strengths", []),
                "weaknesses":       eval_data.get("weaknesses", []),
            }
            interview_results.append(result)

            logger.info(
                "[FLOW] ANSWER EVALUATED current_q=%s evaluated_answers=%d",
                current_q, len(interview_results),
            )
            logger.info(
                "[ANSWER STORED] question_number=%s question=%r candidate_answer=%r",
                current_q, current_question_text, text,
            )

            if current_q < MAX_INTERVIEW_QUESTIONS:
                # Sanity: result count must equal number of questions asked so far
                if len(interview_results) != current_q:
                    logger.error(
                        "[FLOW] CONSISTENCY ERROR results=%d current_q=%d",
                        len(interview_results), current_q,
                    )
                logger.info(
                    "[FLOW] NEXT QUESTION AUTHORIZED current_q=%s evaluated_answers=%d",
                    current_q, len(interview_results),
                )
                interview_phase = "GENERATING_QUESTION"
                await generate_interview_question("after_answer")
            else:
                # Q5 complete â€” no Q6, ever
                interview_phase = "WAITING_FOR_COMPLETION"
                logger.info("[FLOW] Q5 ANSWER EVALUATED")
                logger.info("[FLOW] NO Q6")
                logger.info("[FLOW] WAITING FOR COMPLETE INTERVIEW BUTTON")

                await publish({
                    "type":            "interview_ready_to_complete",
                    "question_number": current_q,
                    "question_count":  question_count,
                })

        except Exception:
            logger.exception("Error in handle_user_speech")
            interview_phase = "WAITING_FOR_ANSWER"  # let user retry
        finally:
            is_processing_turn = False
            logger.info("[PROCESSING LOCK] is_processing_turn = False")
